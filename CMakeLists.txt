cmake_minimum_required(VERSION 3.16)
project(binforge VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(BINFORGE_SOURCES src/base.cc src/reader.cc src/address_space.cc src/parser.cc src/elf_parser.cc src/pe_parser.cc src/macho_parser.cc src/loader.cc src/validator.cc src/disassembler.cc src/disassembly_analyzer.cc src/debug_info.cc src/image_analyzer.cc src/symbol_index.cc src/relocation_decoder.cc src/layout.cc)
add_library(binforge ${BINFORGE_SOURCES})
target_include_directories(binforge PUBLIC include)
if(MSVC)
 target_compile_options(binforge PRIVATE /W4 /permissive-)
else()
 target_compile_options(binforge PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(binforge_cli tools/binforge.cc)
target_link_libraries(binforge_cli PRIVATE binforge)
enable_testing()
add_executable(binforge_tests tests/binforge_tests.cc)
target_link_libraries(binforge_tests PRIVATE binforge)
add_test(NAME binforge_tests COMMAND binforge_tests)
